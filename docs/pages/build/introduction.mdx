  const currentProfile = visibleProfiles[0];

  function nextProfile() {
    if (!currentProfile) return;

    setCurrentIndex(previous => previous + 1);
  }

  function handlePass() {
    if (!currentProfile) return;

    setPassed(previous => [...previous, currentProfile.id]);
  }

  function handleLike() {
    if (!currentProfile) return;

    const person = currentProfile;

    setLiked(previous => [...previous, person.id]);

    if (person.likedYou) {
      setMatches(previous => {
        if (previous.some(p => p.id === person.id)) {
          return previous;
        }
        return [...previous, person];
      });

      Alert.alert(
        "It's a Match! ❤️",
        `You and ${person.name} liked each other!`
      );
    }
  }

  function likeBack(person) {
    setLiked(previous =>
      previous.includes(person.id)
        ? previous
        : [...previous, person.id]
    );

    setMatches(previous => {
      if (previous.some(p => p.id === person.id)) {
        return previous;
      }
      return [...previous, person];
    });

    Alert.alert("It's a Match! ❤️", `You matched with ${person.name}!`);
  }

  function openChat(person) {
    setSelectedChat(person);

  function sendMessage() {
    if (!message.trim() || !selectedChat) return;

      id: Date.now().toString(),
      text: message.trim(),
      sender: "me"
    };

    setChatMessages(previous => ({
      ...previous,
      [selectedChat.id]: [
        ...(previous[selectedChat.id] || []),
        newMessage
      ]
    }));

    setMessage("");
  }
